name: Build APK and website

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write
  pages: write
  id-token: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: "8.7"
      - name: Build APK
        run: gradle assembleRelease --no-daemon
      - name: Name the APK
        run: cp app/build/outputs/apk/release/app-release.apk hulma-3d.apk
      - uses: actions/upload-artifact@v4
        with:
          name: hulma-3d-apk
          path: hulma-3d.apk
      # Makes the QR download link work:
      # https://github.com/<you>/hulma-3d/releases/latest/download/hulma-3d.apk
      - name: Publish release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Hulma 3D build ${{ github.run_number }}
          files: hulma-3d.apk
          make_latest: true

  # Puts the app at https://<you>.github.io/hulma-3d/ (the APK opens this page).
  # Needs Settings > Pages > Source: GitHub Actions (one time).
  website:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deploy.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/upload-pages-artifact@v3
        with:
          path: web
      - id: deploy
        uses: actions/deploy-pages@v4
